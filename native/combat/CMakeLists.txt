cmake_minimum_required(VERSION 3.20)
project(s2s_combat_core LANGUAGES CXX)

# The game-independent core of the combat SFSE plugin (docs/armor/combat-plugin-spec.md). The SFSE plugin itself
# (CommonLibSF, Windows only) links this library; this file builds the library and its tests on any platform.
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_library(s2s_combat STATIC src/combat.cpp src/controller.cpp src/clash.cpp src/left_blade.cpp)
target_include_directories(s2s_combat PUBLIC include)
if(MSVC)
    target_compile_options(s2s_combat PRIVATE /W4 /permissive-)
else()
    target_compile_options(s2s_combat PRIVATE -Wall -Wextra -Wpedantic)
endif()

enable_testing()
add_executable(test_combat tests/test_combat.cpp)
target_link_libraries(test_combat PRIVATE s2s_combat)
add_test(NAME combat_core COMMAND test_combat)
add_executable(test_controller tests/test_controller.cpp)
target_link_libraries(test_controller PRIVATE s2s_combat)
target_compile_definitions(test_controller PRIVATE S2S_SAMPLE_INI="${CMAKE_CURRENT_SOURCE_DIR}/sfse/S2SCombat.ini")
add_test(NAME combat_controller COMMAND test_controller)
add_executable(test_clash tests/test_clash.cpp)
target_link_libraries(test_clash PRIVATE s2s_combat)
target_compile_definitions(test_clash PRIVATE S2S_SAMPLE_INI="${CMAKE_CURRENT_SOURCE_DIR}/sfse/S2SClash.ini")
add_test(NAME combat_clash COMMAND test_clash)
add_executable(test_left_blade tests/test_left_blade.cpp)
target_link_libraries(test_left_blade PRIVATE s2s_combat)
add_test(NAME combat_left_blade COMMAND test_left_blade)
